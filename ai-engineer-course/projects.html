Итоговая система

Идея простая: вместо четырёх учебных репозиториев — одна система, которая растёт неделя за неделей. К концу у вас есть сервис, который отвечает на вопросы по документации с цитатами, измеряет своё качество на каждом PR, доступен агентам через MCP, проверен набором атак и выглядит как продакшн: трейсы, fallback, кэш, SLO, откат флагом, дашборд стоимости. Именно такой проект проходит deep dive на собеседовании: в нём есть решения, цифры и ошибки. В капстоуне (модуль 15) эта же система становится продакшн-агентом с приёмкой по метрикам и защитой design doc.

A · RAG по документации на Go + pgvector

недели 1–6, 14модули 0–5, 11Go, PostgreSQL, pgvector

Цель. Сервис, который индексирует документацию и отвечает на вопросы по ней с цитатами, стримингом и честным «не знаю», если в документации ответа нет. Всё на Go, хранилище — Postgres с pgvector: одна база для чанков, метаданных, полнотекстового индекса и позже кэша. Проект растёт вместе с модулями 0–5: сначала каркас и вызов модели, потом промпты как код, потом retrieval. В неделю 14 к нему возвращаетесь с модулем 11: PDF, PII-редакция, дедупликация и версии корпуса.

CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE chunks (
    id         bigserial PRIMARY KEY,
    doc_url    text NOT NULL,
    ord        int  NOT NULL,
    heading    text NOT NULL DEFAULT '',
    content    text NOT NULL,
    tokens     int  NOT NULL,
    embedding  vector(1024) NOT NULL,
    tsv        tsvector GENERATED ALWAYS AS
               (to_tsvector('simple', heading || ' ' || content)) STORED,
    updated_at timestamptz NOT NULL DEFAULT now(),
    UNIQUE (doc_url, ord)
);
CREATE INDEX chunks_hnsw ON chunks USING hnsw (embedding vector_cosine_ops)
    WITH (m = 16, ef_construction = 64);
CREATE INDEX chunks_tsv ON chunks USING gin (tsv);

-- Гибридный поиск: вектор и полнотекст параллельно, слияние RRF (k = 60).
-- $1 — эмбеддинг запроса, $2 — текст запроса, $3 — сколько вернуть.
WITH v AS (
    SELECT id, row_number() OVER (ORDER BY embedding <=> $1) AS r
    FROM chunks ORDER BY embedding <=> $1 LIMIT 40
), k AS (
    SELECT id, row_number() OVER (ORDER BY ts_rank_cd(tsv, q) DESC) AS r
    FROM chunks, plainto_tsquery('simple', $2) q
    WHERE tsv @@ q
    ORDER BY ts_rank_cd(tsv, q) DESC LIMIT 40
)
SELECT c.id, c.doc_url, c.heading, c.content,
       coalesce(1.0 / (60 + v.r), 0) + coalesce(1.0 / (60 + k.r), 0) AS score
FROM v FULL OUTER JOIN k USING (id)
JOIN chunks c USING (id)
ORDER BY score DESC
LIMIT $3;
// POST /ask {"q": "..."} → text/event-stream: sources, delta*, done | error
func (s *Server) handleAsk(w http.ResponseWriter, r *http.Request) {
	var req struct{ Q string `json:"q"` }
	if err := json.NewDecoder(http.MaxBytesReader(w, r.Body, 8<<10)).Decode(&req); err != nil || req.Q == "" {
		http.Error(w, `{"error":"q required"}`, http.StatusBadRequest)
		return
	}
	ctx := r.Context() // клиент ушёл → ctx отменён → стрим к модели закрывается
	hits, err := s.retriever.Search(ctx, req.Q, 8)
	if err != nil {
		s.log.ErrorContext(ctx, "search", "err", err)
		http.Error(w, `{"error":"search failed"}`, http.StatusBadGateway)
		return
	}
	flusher, ok := w.(http.Flusher)
	if !ok {
		http.Error(w, "streaming unsupported", http.StatusInternalServerError)
		return
	}
	w.Header().Set("Content-Type", "text/event-stream")
	w.Header().Set("Cache-Control", "no-cache")
	w.Header().Set("X-Accel-Buffering", "no") // иначе nginx буферизует стрим
	send := func(event string, v any) {
		b, _ := json.Marshal(v)
		fmt.Fprintf(w, "event: %s\ndata: %s\n\n", event, b)
		flusher.Flush()
	}
	send("sources", hits)
	if len(hits) == 0 || hits[0].Score < s.minScore {
		send("done", map[string]string{"answer": "В документации нет ответа на этот вопрос."})
		return // не генерируем без источников
	}

	stream := s.llm.Messages.NewStreaming(ctx, anthropic.MessageNewParams{
		Model:     "claude-sonnet-5",
		MaxTokens: 2048,
		System: []anthropic.TextBlockParam{{
			Text:         s.systemPrompt, // стабильный префикс → prompt caching
			CacheControl: anthropic.NewCacheControlEphemeralParam(),
		}},
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(buildPrompt(req.Q, hits))),
		},
	})
	defer stream.Close()
	for stream.Next() {
		if d, ok := stream.Current().AsAny().(anthropic.ContentBlockDeltaEvent); ok {
			if t, ok := d.Delta.AsAny().(anthropic.TextDelta); ok {
				send("delta", map[string]string{"text": t.Text})
			}
		}
	}
	if err := stream.Err(); err != nil {
		s.log.ErrorContext(ctx, "llm stream", "err", err)
		send("error", map[string]string{"error": "generation failed"})
		return
	}
	send("done", map[string]any{})
}

import json, sys, statistics, httpx

def recall_at_k(got: list[str], expected: set[str], k: int) -> float:
    return len(set(got[:k]) & expected) / len(expected) if expected else 1.0

def mrr(got: list[str], expected: set[str]) -> float:
    return next((1 / (i + 1) for i, u in enumerate(got) if u in expected), 0.0)

def run(base_url: str, path: str = "data/golden.jsonl") -> dict:
    rows = [json.loads(l) for l in open(path, encoding="utf-8")]
    r5, rr = [], []
    with httpx.Client(base_url=base_url, timeout=30) as c:
        for row in rows:
            if not row["answerable"]:
                continue  # вопросы без ответа проверяются отдельно: ждём «не знаю»
            resp = c.get("/search", params={"q": row["question"], "k": 10})
            resp.raise_for_status()
            urls = [h["doc_url"] for h in resp.json()]
            exp = set(row["expected_urls"])
            r5.append(recall_at_k(urls, exp, 5)); rr.append(mrr(urls, exp))
    return {"recall@5": statistics.mean(r5), "mrr": statistics.mean(rr), "n": len(r5)}

def gate(cur: dict, base: dict, tol: float = 0.02) -> list[str]:
    return [f"{m}: {cur[m]:.3f} < {base[m]:.3f} - {tol}"
            for m in ("recall@5", "mrr") if cur[m] < base[m] - tol]

if __name__ == "__main__":
    cur = run(sys.argv[1])
    fails = gate(cur, json.load(open("baseline.json")))
    print(json.dumps(cur, indent=2))
    if fails:
        print("REGRESSION:\n" + "\n".join(fails)); sys.exit(1)

C · Агент, MCP-сервер и безопасность

недели 7–9, 11модули 6, 7, 8Go, MCP go-sdk, Claude Code/Desktop

Цель. Дать агентам доступ к вашей системе через стандартный протокол. MCP-сервер на Go с инструментами поиска по документации, чтения чанка целиком и чтения из БД (только чтение, allowlist запросов), подключённый к Claude Code и Claude Desktop. Плюс собственный агентный цикл на Go поверх тех же инструментов — чтобы показать, что вы понимаете, что происходит внутри клиента, — с лимитами, компакцией и заметками для длинных задач. Затем, с модулем 7, агент получает threat model, guardrails и набор атак с метрикой ASR (attack success rate, доля успешных атак), а с модулем 8 — eval по траекториям.

// cmd/docs-mcp/main.go — MCP-сервер на официальном Go SDK.
package main

import (
	"context"
	"log"
	"os"

	"github.com/modelcontextprotocol/go-sdk/mcp"

	"example.com/docrag/internal/retrieve"
)

type SearchIn struct {
	Query string `json:"query" jsonschema:"вопрос или ключевые слова на естественном языке"`
	K     int    `json:"k,omitempty" jsonschema:"сколько фрагментов вернуть, 1-10, по умолчанию 5"`
}

type SearchOut struct {
	Results []retrieve.Hit `json:"results"`
}

func main() {
	log.SetOutput(os.Stderr) // stdout занят JSON-RPC: любой лишний вывод ломает протокол
	r, err := retrieve.New(context.Background(), os.Getenv("DATABASE_URL"))
	if err != nil {
		log.Fatal(err)
	}
	server := mcp.NewServer(&mcp.Implementation{Name: "docrag", Version: "0.3.0"}, nil)

	mcp.AddTool(server, &mcp.Tool{
		Name:        "search_docs",
		Description: "Гибридный поиск по документации. Возвращает фрагменты с URL и скором. Используй перед ответом на вопрос о документации.",
	}, func(ctx context.Context, _ *mcp.CallToolRequest, in SearchIn) (*mcp.CallToolResult, SearchOut, error) {
		if in.K <= 0 || in.K > 10 {
			in.K = 5
		}
		hits, err := r.Search(ctx, in.Query, in.K)
		if err != nil {
			return nil, SearchOut{}, err // SDK вернёт клиенту результат с isError
		}
		return nil, SearchOut{Results: hits}, nil
	})

	if err := server.Run(context.Background(), &mcp.StdioTransport{}); err != nil {
		log.Fatal(err)
	}
}

Для Claude Desktop сервер прописывается в его конфигурационном файле в секции mcpServers (команда, аргументы, переменные окружения); путь к файлу и формат смотрите в документации MCP. Для удалённого доступа тот же сервер отдаётся через Streamable HTTP-хендлер из SDK за вашей аутентификацией.

Инструмент чтения из БДНе давайте агенту произвольный SQL, даже «только SELECT». Если хочется свободного SQL — это разбор 3 из модуля 12 со всеми его защитами.

// Спан на вызов модели с атрибутами по семантическим конвенциям OpenTelemetry GenAI.
var tracer = otel.Tracer("docrag/llm")

func (c *Client) Generate(ctx context.Context, p anthropic.MessageNewParams) (*anthropic.Message, error) {
	ctx, span := tracer.Start(ctx, "chat "+string(p.Model), trace.WithSpanKind(trace.SpanKindClient))
	defer span.End()
	span.SetAttributes(
		attribute.String("gen_ai.operation.name", "chat"),
		attribute.String("gen_ai.provider.name", "anthropic"),
		attribute.String("gen_ai.request.model", string(p.Model)),
		attribute.Int64("gen_ai.request.max_tokens", p.MaxTokens),
		attribute.String("app.prompt_version", c.promptVersion),
	)
	start := time.Now()
	msg, err := c.sdk.Messages.New(ctx, p)
	c.latency.Record(ctx, time.Since(start).Seconds(),
		metric.WithAttributes(attribute.String("model", string(p.Model)), attribute.Bool("error", err != nil)))
	if err != nil {
		span.RecordError(err)
		span.SetStatus(codes.Error, "llm call failed")
		return nil, err
	}
	u := msg.Usage
	usd := c.prices.Cost(p.Model, u.InputTokens, u.CacheReadInputTokens, u.OutputTokens)
	span.SetAttributes(
		attribute.Int64("gen_ai.usage.input_tokens", u.InputTokens),
		attribute.Int64("gen_ai.usage.output_tokens", u.OutputTokens),
		attribute.Int64("app.cache_read_input_tokens", u.CacheReadInputTokens),
		attribute.StringSlice("gen_ai.response.finish_reasons", []string{string(msg.StopReason)}),
		attribute.Float64("app.cost_usd", usd),
	)
	c.costUSD.Add(ctx, usd, metric.WithAttributes(attribute.String("model", string(p.Model))))
	return msg, nil
}

Имена атрибутов gen_ai.* берите из текущей версии семантических конвенций OpenTelemetry для GenAI (они ещё развиваются); собственные поля помечайте префиксом приложения, как app.cost_usd.

Самопроверка

Почему golden set появляется уже на неделе 6, хотя модуль Evals — восьмой?

Все последующие изменения (rerank, чанкинг, fallback на другую модель, семантический кэш, guardrails) нужно оценивать числами. Для метрик retrieval хватает модуля 5, поэтому набор собирается сразу после RAG, а judge, статистика и гейт в CI добавляются с модулем 8. Без набора вы не знаете, помогло ли изменение, и на собеседовании нечего показать, кроме «стало лучше».

Почему red-team встроен в проект C, а не вынесен в отдельный проект?

Атаки направлены на конкретную систему: агента с инструментами и данными из A и C. Набор атак — это eval-датасет, который живёт рядом с агентными задачами и гоняется тем же CI. Отдельный проект создал бы иллюзию, что безопасность можно «доделать потом», а не проверять на каждом PR.

Почему в проекте A достаточно pgvector?

Корпус 500–5 000 страниц — это десятки тысяч чанков: HNSW в Postgres отвечает за миллисекунды, а фильтры, полнотекст, метаданные и кэш живут в одной транзакционной базе. Отдельная векторная БД добавила бы эксплуатационную сложность без выигрыша на таком объёме.

Что сломается, если MCP-сервер на stdio пишет логи в stdout?

stdout — канал JSON-RPC; любая лишняя строка будет воспринята клиентом как невалидное сообщение, и соединение сломается. Логи — только в stderr или файл.

Почему инструмент db_query принимает имя запроса, а не SQL?

Чтобы поверхность атаки была конечной: агент может выполнить только заранее описанные и проверенные запросы с параметрами. Свободный SQL требует AST-валидации, проверки прав, EXPLAIN-гейта и других защит, которые в этом проекте не нужны.

Как подобрать порог семантического кэша?

На golden set: для пар вопросов с одинаковым и разным правильным ответом посчитать сходство эмбеддингов и выбрать порог, при котором нет ни одного неверного хита (или их доля ниже допустимой). Затем мониторить в проде: выборочно проверять хиты судьёй.

Какие атрибуты должны быть у спана вызова модели?

Модель, провайдер, операция, max_tokens, токены входа/выхода и cache read, finish reason, стоимость, версия промпта; плюс статус ошибки. Этого достаточно, чтобы построить дашборд стоимости и найти причину плохого ответа.

Вопросы с собеседований по проектам

Расскажите про архитектуру вашего RAG-проекта. Почему такие решения?

Три-пять компонентов с причинами: структурный чанкинг (эксперимент показал +N recall против фиксированного размера), pgvector (объём, фильтры и полнотекст в одном запросе; переезд при десятках миллионов векторов или высоком QPS), гибрид + RRF (идентификаторы в документации), порог уверенности (не генерируем без источников). Каждое решение — с альтернативой и цифрой.

Как вы поняли, что система работает хорошо?

Golden set из N вопросов с источниками: какие типы, как собран. Retrieval: recall@5 X → Y. Генерация: faithfulness Z по судье, откалиброванному на моей разметке (согласие W%). Гейт в CI. И ограничения: набор маленький, различия меньше 2–3 п.п. считаю шумом.

Что было самой большой ошибкой в проекте?

Конкретный случай с тем, как нашли и какие цифры до и после: например, чанкинг по символам рвал код-блоки, recall был 0.48; нашёл через разбор худших примеров в отчёте evals; структурный чанкинг дал 0.7+. Вывод: сначала смотреть на данные, потом тюнить модель.

Что произойдёт с вашим сервисом, если Anthropic API недоступен 10 минут?

Circuit breaker откроется после серии ошибок, запросы пойдут на резервную модель; если недоступно всё — API отдаёт найденные фрагменты с сообщением, что генерация временно недоступна. Кэш отвечает на популярные вопросы. Это покрыто chaos-тестом с фейковым провайдером; на дашборде видна доля fallback.

Сколько стоит ваш сервис и как вы это посчитали?

Из трейсов: средние токены входа/выхода и доля cache read, умножить на цены. Например, $0.003 за вопрос; при 1 000 вопросах в день ≈ $90 в месяц. Главные рычаги: prompt caching системного промпта, меньше чанков после rerank, кэш ответов. Индексация — разовая стоимость эмбеддингов.

Зачем вам MCP, если есть HTTP API?

HTTP API — для моего фронтенда и интеграций, которые я пишу сам. MCP — стандартный способ дать инструменты любому агентному клиенту (Claude Code, Desktop, IDE) без написания интеграции под каждый: клиент сам обнаруживает инструменты и их схемы. Цена — ещё один интерфейс для поддержки и отдельная модель безопасности (кто может подключиться, что могут инструменты).

Как вы проверяли, что агента нельзя заставить слить данные через документ или результат инструмента?

Начал с threat model: какие приватные данные видит агент, откуда приходит недоверенный текст, какие есть каналы наружу. Затем набор из N атак как eval-датасет: прямые и непрямые инъекции через документы корпуса и результаты инструментов, утечка system prompt, эксфильтрация через markdown-картинки. ASR до защит — X%, после — Y%, с интервалом. Главное снижение дала не инструкция в промпте, а архитектура: инструменты только на чтение и с правами пользователя, фильтр ссылок на внешние домены на выходе. Набор гоняется в CI, новая успешная атака валит PR.

На что смотрит интервьюер: есть ли измерение, а не «добавили guardrails»; понимает ли кандидат, что промптом инъекцию не закрыть; какие ноги lethal trifecta разорваны детерминированно.

Как бы вы масштабировали систему до 10M документов и 100 RPS?

Посчитать: 10M документов ≈ 80M чанков, при d=1024 ≈ 330 ГБ float32 — не влезает в один Postgres; квантизация (halfvec/int8/binary в pgvector) или выделенная векторная БД с шардированием. Ingest — очередь и воркеры. Генерация: 100 RPS упираются в rate limits провайдера — тиры, несколько провайдеров через gateway, кэш. Эмбеддинг запросов — свой сервис рядом с поиском. Evals остаются тем же гейтом.
